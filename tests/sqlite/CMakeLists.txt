cmake_minimum_required(VERSION 3.24)
project(ShrootSQLiteIntegration LANGUAGES C ASM)
set(root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(SQLITE_AMALGAMATION "" CACHE PATH "External SQLite source directory")
set(SQLITE_KIND STATIC CACHE STRING "External target kind: STATIC or SHARED")
set(SQLITE_FEATURES "SQLITE_ENABLE_PREUPDATE_HOOK;SQLITE_THREADSAFE=2" CACHE STRING "External SQLite compilation flags")
option(TEST_RUNTIME "Build the full runtime using this provider" OFF)
option(TEST_REJECTION "Expect SQLite admission to reject this provider" OFF)
if(SQLITE_AMALGAMATION)
    add_library(integrator-sqlite ${SQLITE_KIND} "${SQLITE_AMALGAMATION}/sqlite3.c")
    target_include_directories(integrator-sqlite PUBLIC "${SQLITE_AMALGAMATION}")
    target_compile_definitions(integrator-sqlite PRIVATE ${SQLITE_FEATURES})
    target_compile_options(integrator-sqlite PRIVATE -O2)
    set(SHROOT_SQLITE integrator-sqlite)
endif()
if(TEST_RUNTIME)
    add_subdirectory("${root}" runtime)
    if(SQLITE_AMALGAMATION AND SQLITE_KIND STREQUAL "SHARED")
        # The embedding project, not Shroot, packages and locates its provider.
        install(TARGETS integrator-sqlite LIBRARY DESTINATION lib COMPONENT Shroot)
        set_target_properties(supervisor service image PROPERTIES INSTALL_RPATH "$ORIGIN/../../lib")
    endif()
else()
    enable_testing()
    include("${root}/cmake/SQLite.cmake")
    add_executable(test_database ../test_storage_provider.c ../test_storage_database.c "${root}/src/private_tree.c")
    target_include_directories(test_database PRIVATE "${root}/src")
    target_compile_options(test_database PRIVATE -std=c17 -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(test_database PRIVATE guest-database Threads::Threads)
    if(TEST_REJECTION)
        add_test(NAME database COMMAND test_database --reject)
    else()
        add_test(NAME database COMMAND test_database)
    endif()
endif()
