function(guest_executable name)
    cmake_parse_arguments(guest "HOST_LIBC" "" "" ${ARGN})
    set(code ${shroot_native_code})
    set(link ${shroot_native_link})
    if(guest_HOST_LIBC)
        # Host utilities need the platform resolver and its current network.
        # In particular, static Bionic does not initialize the netd client.
        set(code -fPIE)
        set(link -pie)
    endif()
    add_executable(${name} ${guest_UNPARSED_ARGUMENTS})
    set_target_properties(${name} PROPERTIES OUTPUT_NAME libmagicdesk_guest_${name}.so)
    target_compile_options(${name} PRIVATE -std=c17 -O2 -Wall -Wextra -Werror ${code} -fno-builtin)
    target_link_options(${name} PRIVATE ${link} -Wl,-z,noexecstack
        -Wl,-z,relro,-z,now -Wl,-z,max-page-size=16384)
    target_link_libraries(${name} PRIVATE guest-parcels guest-objects guest-svc guest-interruptible guest-seccomp Threads::Threads)
    list(FIND ARGN "${src}/network_stack.c" packet_stack)
    if(NOT packet_stack EQUAL -1)
        target_link_libraries(${name} PRIVATE guest-lwip)
    endif()
endfunction()
function(guest_freestanding_code name)
    # NDK defaults assume libc; freestanding helpers use only their own raw syscall gate.
    target_compile_options(${name} PRIVATE -ffreestanding -U_FORTIFY_SOURCE
        -fno-stack-protector -fvisibility=hidden)
endfunction()
function(guest_freestanding name)
    guest_freestanding_code(${name})
    target_compile_options(${name} PRIVATE -fno-pie)
    # These images have their own entry and layout, never a host dynamic linker.
    set_property(TARGET ${name} PROPERTY LINK_OPTIONS -static -nostdlib
        -Wl,-z,defs -Wl,-z,noexecstack -Wl,-z,max-page-size=16384)
endfunction()
